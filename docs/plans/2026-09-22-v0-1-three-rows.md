```ts
import type { ActionId, ItemId, SkillId } from '../data/types';

/** One mastery ledger. `exp` is progress toward the NEXT level. */
export interface Ledger { readonly level: number; readonly exp: number }

/** MECHANICS section 3: two ledgers advancing from the same effort. */
export interface SkillState { readonly core: Ledger; readonly run: Ledger }

export interface QueueEntry {
  readonly actionId: ActionId;
  readonly progress: number;
  readonly costsConsumed: number;
  /** Remaining completions; absent means repeat until removed. */
  readonly targetCount?: number;
  /** The count as queued, for display as done/total. Set with targetCount. */
  readonly targetTotal?: number;
  /** Waiting on an input it cannot pay. It stays in the queue (spec 8.5, 9). */
  readonly stalled: boolean;
}

export type PauseReason = 'none' | 'player' | 'system';

/** What happened this tick, for the UI to narrate. The engine writes no prose. */
export type GameEvent =
  | { readonly type: 'stalled'; readonly actionId: ActionId; readonly item: ItemId }
  | { readonly type: 'resumed'; readonly actionId: ActionId }
  | { readonly type: 'completed'; readonly actionId: ActionId; readonly oneTime: boolean }
  | { readonly type: 'coreLevel'; readonly skill: SkillId; readonly level: number }
  | { readonly type: 'died'; readonly runTicks: number };

export interface GameState {
  readonly runTicks: number;
  readonly health: number;
  readonly maxHealth: number;
  readonly paused: PauseReason;
  readonly dead: boolean;
  readonly skills: Readonly<Record<SkillId, SkillState>>;
  readonly inventory: Readonly<Record<ItemId, number>>;
  /** Ticks until each food may be eaten again. Absent means ready. */
  readonly foodCooldowns: Readonly<Record<ItemId, number>>;
  readonly queue: readonly QueueEntry[];
  readonly completedOneTime: readonly ActionId[];
  /** Lifetime completions by templateKey. Drives automation later. */
  readonly completionCounts: Readonly<Record<string, number>>;
  readonly decayMultiplier: number;
  /** This tick's events. Replaced every tick; never accumulates. */
  readonly events: readonly GameEvent[];
```

```ts
// src/engine/skills.ts
/**
 * Dual mastery, MECHANICS section 3. Two ledgers advance from one amount of
 * effort per tick; each converts it to levels at its own rate. Pure.
 */
import { balance } from '../balance';
import type { Ledger, SkillState } from './types';

const { coreMastery, runMastery, expCurveExponent, baseTickExp } = balance.skills;

export function newSkill(): SkillState {
  return { core: { level: 0, exp: 0 }, run: { level: 0, exp: 0 } };
}

/** XP a ledger needs to go from `level` to `level + 1`. */
export function expToNextLevel(baseExp: number, level: number): number {
  return baseExp * Math.pow(expCurveExponent, level);
}

/** (1 + core x per-level) x (1 + run x per-level) x tool. */
export function multiplier(s: SkillState, tool = 1): number {
  return (1 + s.core.level * coreMastery.multiplierPerLevel)
    * (1 + s.run.level * runMastery.multiplierPerLevel)
    * tool;
}

/** XP earned by this skill in one tick of running its action. */
export function tickExp(s: SkillState, tool = 1): number {
  return baseTickExp * multiplier(s, tool);
}

function advance(ledger: Ledger, baseExp: number, exp: number): { ledger: Ledger; gained: number } {
  let level = ledger.level;
  let pool = ledger.exp + exp;
  let cost = expToNextLevel(baseExp, level);
  let gained = 0;
  while (pool >= cost) {
    pool -= cost;
    level += 1;
    gained += 1;
    cost = expToNextLevel(baseExp, level);
  }
  return { ledger: { level, exp: pool }, gained };
}

/** Add one tick's effort to BOTH ledgers. Not a split: the same amount twice. */
export function award(s: SkillState, exp: number): { skill: SkillState; coreLevelsGained: number } {
  const core = advance(s.core, coreMastery.baseExp, exp);
```

```ts
// src/engine/inventory.ts
/**
 * The pack. Counts are plain numbers keyed by item id; caps come from content
 * (spec section 9: per-item stack caps, and production past the cap is lost).
 * Pure.
 */
import type { Content, ItemId } from '../data/types';

export type Inventory = Readonly<Record<ItemId, number>>;

export function count(inv: Inventory, id: ItemId): number {
  return inv[id] ?? 0;
}

/** Add up to the cap. Reports how many actually landed so a full stack is visible. */
export function add(inv: Inventory, content: Content, id: ItemId, n: number): { inventory: Inventory; added: number } {
  const cap = content.items[id]?.cap ?? Number.POSITIVE_INFINITY;
  const have = count(inv, id);
  const added = Math.max(0, Math.min(n, cap - have));
  return { inventory: { ...inv, [id]: have + added }, added };
}

export function has(inv: Inventory, id: ItemId, n: number): boolean {
  return count(inv, id) >= n;
}

export function take(inv: Inventory, id: ItemId, n: number): Inventory {
  const have = count(inv, id);
  if (have < n) throw new Error(`take: ${id} has ${have}, need ${n}`);
  return { ...inv, [id]: have - n };
}
```

```ts
// src/engine/costs.ts
/**
 * Incremental cost consumption, MECHANICS section 2: costs are paid one unit
 * at a time as progress accrues, never as a lump sum on completion. Declaration
 * order is spend order.
 */
import type { ActionDefinition, ItemId } from '../data/types';

export function totalUnits(action: ActionDefinition): number {
  return action.itemCosts.reduce((sum, c) => sum + c.amount, 0);
}

/** How many units must have been consumed at this progress. One is spent to begin. */
export function requiredCostsConsumed(action: ActionDefinition, progress: number): number {
  const total = totalUnits(action);
  if (total === 0) return 0;
  const expPerUnit = action.expCost / total;
  return Math.min(total, Math.floor(progress / expPerUnit) + 1);
}

/** The item the next unit comes from, walking itemCosts in declared order. */
export function nextCostItem(action: ActionDefinition, costsConsumed: number): ItemId | null {
  let seen = 0;
  for (const c of action.itemCosts) {
    if (costsConsumed < seen + c.amount) return c.item;
    seen += c.amount;
  }
  return null;
}
```

```ts
// src/engine/queue.ts
/**
 * The action queue, MECHANICS section 2, with the changes the spec declares in
 * section 9: a waiting entry stays in the queue, the engine works the first
 * entry that can run this tick, and a flagged entry is cleared the moment its
 * input is there. Production past a stack's cap is lost. Nothing is re-paid.
 * The engine reports what happened as typed events and writes no prose.
 * No automation here.
 */
import { balance } from '../balance';
import type { ActionDefinition, ActionId, Content, ItemId, SkillId } from '../data/types';
import { SKILL_IDS } from '../data/types';
import type { GameEvent, GameState, QueueEntry, SkillState } from './types';
import { award, newSkill, tickExp } from './skills';
import { add, has, take } from './inventory';
import { nextCostItem, requiredCostsConsumed, totalUnits } from './costs';

export function newState(): GameState {
  const skills = Object.fromEntries(SKILL_IDS.map((id) => [id, newSkill()])) as Record<SkillId, SkillState>;
  return {
    runTicks: 0,
    health: balance.health.base,
    maxHealth: balance.health.base,
    paused: 'system',
    dead: false,
    skills,
    inventory: {},
    foodCooldowns: {},
    queue: [],
    completedOneTime: [],
    completionCounts: {},
    decayMultiplier: 1,
    events: [],
  };
}

export function enqueue(
  state: GameState,
  content: Content,
  actionId: ActionId,
  opts: { front?: boolean; targetCount?: number } = {},
): GameState {
  const action = content.actions[actionId];
  if (!action) return state;
  const existing = state.queue.findIndex((e) => e.actionId === actionId);
  if (existing !== -1 && opts.front) {
    const entry = state.queue[existing]!;
    return { ...state, queue: [entry, ...state.queue.filter((_, i) => i !== existing)] };
  }
  if (action.isOneTime && (state.completedOneTime.includes(actionId) || existing !== -1)) return state;
  const entry: QueueEntry = {
    actionId,
    progress: 0,
    costsConsumed: 0,
    stalled: false,
    ...(opts.targetCount !== undefined ? { targetCount: opts.targetCount, targetTotal: opts.targetCount } : {}),
  };
  return { ...state, queue: opts.front ? [entry, ...state.queue] : [...state.queue, entry] };
}

export function removeAt(state: GameState, index: number): GameState {
  return { ...state, queue: state.queue.filter((_, i) => i !== index) };
}

/** The input this entry cannot pay for right now, or null if it can proceed. */
export function missingInput(state: GameState, content: Content, entry: QueueEntry): ItemId | null {
  const action = content.actions[entry.actionId];
  if (!action || entry.costsConsumed >= totalUnits(action)) return null;
  if (requiredCostsConsumed(action, entry.progress) <= entry.costsConsumed) return null;
  const item = nextCostItem(action, entry.costsConsumed);
  return item !== null && !has(state.inventory, item, 1) ? item : null;
}

/** The entry that can do work this tick, flagged or not. -1 if none. */
export function firstRunnable(state: GameState, content: Content): number {
  return state.queue.findIndex((e) => content.actions[e.actionId] !== undefined && missingInput(state, content, e) === null);
}

/** Pay units until costsConsumed matches what progress requires. `short` names the item on a shortfall. */
function payDue(state: GameState, action: ActionDefinition, entry: QueueEntry): { state: GameState; entry: QueueEntry; short?: ItemId } {
  let inventory = state.inventory;
  let consumed = entry.costsConsumed;
  const due = requiredCostsConsumed(action, entry.progress);
  while (consumed < due) {
    const item = nextCostItem(action, consumed);
    if (item === null) break;
    if (!has(inventory, item, 1)) return { state: { ...state, inventory }, entry: { ...entry, costsConsumed: consumed }, short: item };
    inventory = take(inventory, item, 1);
    consumed += 1;
  }
  return { state: { ...state, inventory }, entry: { ...entry, costsConsumed: consumed } };
}

function complete(state: GameState, content: Content, action: ActionDefinition, index: number, events: GameEvent[]): GameState {
  let next = state;
  if (action.producedItem) {
    next = { ...next, inventory: add(next.inventory, content, action.producedItem, action.producedAmount ?? 1).inventory };
  }
  if (action.healthDecayMultiplier !== undefined) {
    next = { ...next, decayMultiplier: next.decayMultiplier * action.healthDecayMultiplier };
  }
  const key = action.templateKey ?? action.id;
  next = { ...next, completionCounts: { ...next.completionCounts, [key]: (next.completionCounts[key] ?? 0) + 1 } };
  const entry = next.queue[index]!;
  const remaining = entry.targetCount === undefined ? undefined : entry.targetCount - 1;
  const done = action.isOneTime || remaining === 0;
  const fresh: QueueEntry = { actionId: entry.actionId, progress: 0, costsConsumed: 0, stalled: false, ...(remaining !== undefined ? { targetCount: remaining, targetTotal: entry.targetTotal } : {}) };
  const queue = done ? next.queue.filter((_, i) => i !== index) : next.queue.map((e, i) => (i === index ? fresh : e));
  events.push({ type: 'completed', actionId: action.id, oneTime: action.isOneTime });
  return { ...next, queue, completedOneTime: action.isOneTime ? [...next.completedOneTime, action.id] : next.completedOneTime };
}

function setEntry(state: GameState, index: number, entry: QueueEntry): GameState {
  return { ...state, queue: state.queue.map((e, i) => (i === index ? entry : e)) };
}

/**
 * MECHANICS tick steps 2 to 9, without automation. The caller has already
 * decided this tick happens (firstRunnable !== -1) and advanced the clock.
 */
export function stepQueue(state: GameState, content: Content): GameState {
  const events: GameEvent[] = [];
  // Clear the flag on every entry that can run again: waiting ends the moment it can.
  let next: GameState = {
    ...state,
    events: [],
    queue: state.queue.map((e) => {
      if (!e.stalled || missingInput(state, content, e) !== null) return e;
      events.push({ type: 'resumed', actionId: e.actionId });
      return { ...e, stalled: false };
    }),
  };
  const index = firstRunnable(next, content);
  if (index === -1) return { ...next, events };
  const entry = next.queue[index]!;
  const action = content.actions[entry.actionId]!;

  // Pay what is due before advancing: the first unit is spent just to begin.
  const paid = payDue(next, action, entry);
  if (paid.short !== undefined) {
    // Unreachable when firstRunnable chose this entry; kept so a future change cannot pay past a shortfall.
    events.push({ type: 'stalled', actionId: action.id, item: paid.short });
    return { ...setEntry(paid.state, index, { ...paid.entry, stalled: true }), events };
  }
  next = paid.state;

  const skill = next.skills[action.verb];
  const gain = tickExp(skill);
  const awarded = award(skill, gain);
  next = { ...next, skills: { ...next.skills, [action.verb]: awarded.skill } };
  for (let l = 0; l < awarded.coreLevelsGained; l++) {
    events.push({ type: 'coreLevel', skill: action.verb, level: awarded.skill.core.level - awarded.coreLevelsGained + l + 1 });
  }
  const advanced: QueueEntry = { ...paid.entry, progress: paid.entry.progress + gain };

  if (advanced.progress >= action.expCost) {
    return { ...complete(setEntry(next, index, advanced), content, action, index, events), events };
  }

  const paidAfter = payDue(next, action, advanced);
  if (paidAfter.short !== undefined) {
    events.push({ type: 'stalled', actionId: action.id, item: paidAfter.short });
```

```ts
// src/engine/health.ts
/**
 * The decay clock and automatic eating, MECHANICS section 4. Decay is a
 * function of ticks elapsed this run; eating is once per tick, per food type,
 * never into overheal, with a cooldown per food. Pure.
 */
import { balance } from '../balance';
import type { Content } from '../data/types';
import { ticksToMinutes } from './time';
import { count, take } from './inventory';
import type { GameState } from './types';

export function damagePerTick(runTicks: number, decayMultiplier: number): number {
  const { baseDamagePerTick, decayGrowthRate } = balance.health;
  return baseDamagePerTick * Math.pow(decayGrowthRate, ticksToMinutes(runTicks)) * decayMultiplier;
}

export function applyDecay(state: GameState): GameState {
  const health = state.health - damagePerTick(state.runTicks, state.decayMultiplier);
  if (health <= 0) {
    return { ...state, health: 0, dead: true, paused: 'system', events: [...state.events, { type: 'died', runTicks: state.runTicks }] };
  }
  return { ...state, health };
}

export function eat(state: GameState, content: Content): GameState {
  let { health, inventory } = state;
  const foodCooldowns: Record<string, number> = { ...state.foodCooldowns };
  for (const item of Object.values(content.items)) {
    if (item.healPerUnit === undefined) continue;
    const cooling = foodCooldowns[item.id] ?? 0;
    if (cooling > 0) { foodCooldowns[item.id] = cooling - 1; continue; }
    if (count(inventory, item.id) === 0) continue;
    if (health + item.healPerUnit > state.maxHealth) continue;
    health += item.healPerUnit;
    inventory = take(inventory, item.id, 1);
    foodCooldowns[item.id] = balance.health.foodCooldownTicks;
  }
  return { ...state, health, inventory, foodCooldowns };
}
```

```ts
// src/engine/tick.ts
/**
 * One tick. Time passes only while work happens (spec section 9, decision B):
 * paused, dead, or nothing that can run this tick, and the tick does not
 * happen at all. Otherwise: advance the clock, decay, eat, work the queue.
 */
import type { Content } from '../data/types';
import { applyDecay, eat } from './health';
import { firstRunnable, stepQueue } from './queue';
import type { GameState, PauseReason } from './types';

export function setPaused(state: GameState, paused: PauseReason): GameState {
  return { ...state, paused };
}

export function step(state: GameState, content: Content): GameState {
  if (state.paused !== 'none' || state.dead || firstRunnable(state, content) === -1) return state;
  let next: GameState = { ...state, runTicks: state.runTicks + 1, events: [] };
  next = applyDecay(next);
  if (next.dead) return next;
  next = eat(next, content);
  const earlier = next.events;
  const worked = stepQueue(next, content);
  return { ...worked, events: [...earlier, ...worked.events] };
}
```

`step` returns the identical object when nothing happens, so React does not re-render on an idle tick.

```ts
// src/state/useGame.ts
/**
 * The only clock in the game. `src/state/` supplies the host capabilities the
 * engine deliberately takes as parameters: here, the interval. The reducer is a
 * thin switch over pure engine functions, plus the log: engine events with the
 * tick they happened on. Words are the UI's job (src/ui/narrate.ts).
 */
import { useEffect, useReducer } from 'react';
import { balance } from '../balance';
import type { ActionId, Content } from '../data/types';
import { enqueue, newState, removeAt } from '../engine/queue';
import { setPaused, step } from '../engine/tick';
import type { GameEvent, GameState } from '../engine/types';

export type GameAction =
  | { type: 'queue'; actionId: ActionId; front?: boolean }
  | { type: 'remove'; index: number }
  | { type: 'pause' }
  | { type: 'resume' }
  | { type: 'tick' };

export type LogEvent = GameEvent | { readonly type: 'lifeBegins'; readonly life: number };

export interface LogLine { readonly at: number; readonly event: LogEvent }

interface Model { readonly state: GameState; readonly log: readonly LogLine[] }

function sameEvent(a: LogEvent, b: LogEvent): boolean {
  return JSON.stringify(a) === JSON.stringify(b);
}

function withLog(model: Model, state: GameState): Model {
  const worth = state.events.filter((e) => !(e.type === 'completed' && !e.oneTime));
  if (worth.length === 0) return { state, log: model.log };
  let log = model.log;
  for (const event of worth) {
    if (log[0] && sameEvent(log[0].event, event)) continue;
    log = [{ at: state.runTicks, event }, ...log];
  }
  return { state, log };
}

function reduce(content: Content) {
  return (model: Model, action: GameAction): Model => {
    const s = model.state;
    switch (action.type) {
      case 'queue': return { ...model, state: enqueue(s, content, action.actionId, { front: action.front ?? false }) };
      case 'remove': return { ...model, state: removeAt(s, action.index) };
      case 'pause': return { ...model, state: setPaused(s, 'player') };
      case 'resume': return { ...model, state: s.dead ? s : setPaused(s, 'none') };
      case 'tick': {
        const next = step(s, content);
        return next === s ? model : withLog(model, next);
      }
    }
  };
}

function initial(): Model {
  return { state: newState(), log: [{ at: 0, event: { type: 'lifeBegins', life: 1 } }] };
}

export function useGame(content: Content): { state: GameState; log: readonly LogLine[]; dispatch: (a: GameAction) => void } {
  const [model, dispatch] = useReducer(reduce(content), undefined, initial);
  useEffect(() => {
    const id = setInterval(() => dispatch({ type: 'tick' }), balance.time.tickIntervalMs);
    return () => clearInterval(id);
  }, []);
  return { state: model.state, log: model.log, dispatch };
}
```

```ts
// src/ui/format.ts
/** Display formatting only. Spec section 8.1: fractions are a/b, no spaces. */
export function fraction(a: number, b: number): string {
  return `${a.toFixed(1)}/${b.toFixed(1)}`;
}

/**
 * Seconds in a minute. A unit conversion nobody may tune, kept out of
 * balance.ts on the same reasoning as MS_PER_SECOND in src/engine/time.ts.
 */
const SECONDS_PER_MINUTE = 60;

/** "7s" under a minute, "m:ss" above. Durations are unpadded (spec 8.1). */
export function seconds(s: number): string {
  const whole = Math.ceil(s);
  if (whole < SECONDS_PER_MINUTE) return `${whole}s`;
  const m = Math.floor(whole / SECONDS_PER_MINUTE);
  const r = whole % SECONDS_PER_MINUTE;
  return `${m}:${String(r).padStart(2, '0')}`;
}

/** The run clock: "mm:ss", padded (spec 8.1). */
export function clock(s: number): string {
  const whole = Math.floor(s);
  const m = Math.floor(whole / SECONDS_PER_MINUTE);
  const r = whole % SECONDS_PER_MINUTE;
  return `${String(m).padStart(2, '0')}:${String(r).padStart(2, '0')}`;
}
```

```ts
// src/ui/narrate.ts
/** The one place engine events become words. Spec 8.6, 9. */
import { SKILLS } from '../data/skills';
import type { ActionId, Content } from '../data/types';
import { ticksToSeconds } from '../engine/time';
import type { LogEvent } from '../state/useGame';
import { clock } from './format';

export interface Narration { readonly kind: 'story' | 'note'; readonly text: string }

function rowName(content: Content, id: ActionId): string {
  const a = content.actions[id];
  return a ? `${SKILLS[a.verb].name} ${a.noun}` : id;
}

export function narrate(e: LogEvent, content: Content): Narration {
  switch (e.type) {
    case 'lifeBegins': return { kind: 'note', text: `Life ${e.life} begins` };
    case 'stalled': return { kind: 'note', text: `${rowName(content, e.actionId)} is waiting on ${e.item}` };
    case 'resumed': return { kind: 'note', text: `${rowName(content, e.actionId)} resumes` };
    case 'completed': {
      const beat = content.actions[e.actionId]?.beat;
      return e.oneTime && beat ? { kind: 'story', text: beat } : { kind: 'note', text: `${rowName(content, e.actionId)} is done` };
    }
    case 'coreLevel': return { kind: 'note', text: `${SKILLS[e.skill].name} reaches Lv ${e.level}` };
    case 'died': return { kind: 'note', text: `Dead at ${clock(ticksToSeconds(e.runTicks))}` };
  }
}
```

```ts
// src/test-utils/realClick.ts
/**
 * A click the way a mouse does it: mousedown, a real gap, mouseup, click, all
 * on the element that was pressed. Synthetic `element.click()` skips the gap,
 * which is exactly where a render loop that rebuilds children makes the browser
 * drop the click (CLAUDE.md gotchas; issue #34). Test-only.
 *
 * Uses a real timer: call it under real timers, and wrap the await in `act`.
 */

/**
 * The gap a real press has, in the middle of the 50-150 ms range CLAUDE.md
 * cites. A test fixture, not a tuning value; it is not in balance.ts for the
 * same reason MS_PER_SECOND is not.
 */
export const REAL_CLICK_GAP_MS = 100;

export async function realClick(el: Element): Promise<void> {
  const opts = { bubbles: true, cancelable: true, view: window };
  el.dispatchEvent(new MouseEvent('mousedown', opts));
  await new Promise((r) => setTimeout(r, REAL_CLICK_GAP_MS));
  el.dispatchEvent(new MouseEvent('mouseup', opts));
  el.dispatchEvent(new MouseEvent('click', opts));
}
```

```tsx
import { useEffect, useState } from 'react';
import { balance } from '../balance';
import { SKILLS } from '../data/skills';
import type { ActionDefinition, Content, ItemId } from '../data/types';
import { count } from '../engine/inventory';
import { tickExp } from '../engine/skills';
import { ticksPerSecond } from '../engine/time';
import type { GameState } from '../engine/types';
import { seconds } from './format';
import { SKILL_ICONS } from './icons';

/**
 * How long the click instruction holds before fading (spec 8.4: "a few
 * seconds"). Presentation timing, not a gameplay value; kept out of balance.ts
 * on the same reasoning as MS_PER_SECOND in src/engine/time.ts.
 */
const INSTRUCTION_MS = 3000;

function makerOf(content: Content, item: ItemId): string {
  const producer = Object.values(content.actions).find((a) => a.producedItem === item);
  return producer ? SKILLS[producer.verb].name : SKILLS.craft.name;
}

export function ActionRow({ action, content, state, running, onNow, onStop, onQueue }: {
  action: ActionDefinition; content: Content; state: GameState; running: boolean;
  onNow: (id: string) => void; onStop: (id: string) => void; onQueue: (id: string) => void;
}) {
  const Icon = SKILL_ICONS[action.verb];
  const perSecond = tickExp(state.skills[action.verb]) * ticksPerSecond();
  const duration = action.expCost / perSecond;
  const shortfalls = action.itemCosts
    .map((c) => ({ ...c, have: count(state.inventory, c.item) }))
    .filter((c) => c.have < c.amount);
  const [instructing, setInstructing] = useState(false);
  useEffect(() => {
    if (!instructing) return;
    const id = setTimeout(() => setInstructing(false), INSTRUCTION_MS);
    return () => clearTimeout(id);
  }, [instructing]);
  const press = (fn: (id: string) => void) => () => { fn(action.id); if (shortfalls.length > 0) setInstructing(true); };
  const key = action.templateKey ?? action.id;
  const done = state.completionCounts[key] ?? 0;
  const needed = action.isOneTime ? balance.automation.unlockOneTime : balance.automation.unlockRepeatable;
  const output = action.producedItem
    ? (action.itemCosts.length === 0 ? `+${action.producedAmount ?? 1} ${action.producedItem}` : action.producedItem)
    : '';

  return (
    <div className={`item row${running ? ' row--on working' : ''}`} data-action={action.id}>
      <div className="row__c1">
        <Icon aria-hidden="true" />
        <span><b>{SKILLS[action.verb].name}</b>{action.noun}{running && <span className="visually-hidden">running</span>}</span>
      </div>
      <div className="row__c2">
        {instructing ? (
          <div className="row__say" role="status">{shortfalls.map((c) => (
            <div key={c.item}>missing {c.amount - c.have} {c.item} · <b>{makerOf(content, c.item)} some!</b></div>
          ))}</div>
        ) : (
          <>
            <div className="row__in">{action.itemCosts.map((c) => {
              const have = count(state.inventory, c.item);
              const short = have < c.amount;
              return <span key={c.item} className={short ? 'row__short' : ''}>{short && '⚠ '}{c.amount} {c.item}{short && <small> have {have}</small>}</span>;
            })}</div>
            <div className="row__arr" aria-hidden="true">→</div>
            <div className="row__out">{output}</div>
          </>
        )}
      </div>
      <div className="row__c3">
        <div className="row__tx">{seconds(duration)}<small>+{action.expCost.toFixed(1)} xp</small></div>
        {running
          ? <button type="button" className="btn btn--stop" aria-label="stop" onClick={() => onStop(action.id)}>■</button>
          : <button type="button" className="btn btn--play" aria-label="do it now" onClick={press(onNow)}>▶</button>}
        <button type="button" className="btn" aria-label="add to queue" onClick={press(onQueue)}>+</button>
        <span className="auto" role="status" aria-label={done === 0 ? 'automation, not yet earned' : `automation, ${done} of ${needed} to earn`}>
          {done > 0 && <>{done}/{needed}<i style={{ width: `${Math.min(100, (done / needed) * 100)}%` }} /></>}
        </span>
      </div>
    </div>
  );
}
```

```tsx
// src/ui/Queue.tsx
import { SKILLS } from '../data/skills';
import type { Content } from '../data/types';
import { count } from '../engine/inventory';
import { firstRunnable, missingInput } from '../engine/queue';
import { tickExp } from '../engine/skills';
import { ticksPerSecond } from '../engine/time';
import type { GameState, QueueEntry } from '../engine/types';
import { seconds } from './format';
import { SKILL_ICONS } from './icons';

function remainingSeconds(state: GameState, content: Content, e: QueueEntry): number {
  const a = content.actions[e.actionId];
  if (!a) return 0;
  const perSecond = tickExp(state.skills[a.verb]) * ticksPerSecond();
  const thisOne = (a.expCost - e.progress) / perSecond;
  const more = e.targetCount === undefined ? 0 : Math.max(0, e.targetCount - 1) * (a.expCost / perSecond);
  return thisOne + more;
}

export function Queue({ state, content, onRemove }: { state: GameState; content: Content; onRemove: (i: number) => void }) {
  const working = firstRunnable(state, content);
  const runnable = state.queue.filter((e) => missingInput(state, content, e) === null);
  const total = runnable.reduce((sum, e) => sum + remainingSeconds(state, content, e), 0);
  const waiting = state.queue.length - runnable.length;
  return (
    <section className="queue" aria-label="queue">

        const pct = Math.min(100, (e.progress / a.expCost) * 100);

            <div className="entry__bar">
              <div className="bar" aria-hidden="true"><div className={`bar__fill${missing ? '' : ' bar__fill--run'}`} style={{ width: `${pct}%` }} /></div>
              <div className="bar__value">{e.targetTotal !== undefined && e.targetCount !== undefined ? `${e.targetTotal - e.targetCount}/${e.targetTotal}` : ''}</div>
            </div>
            <div className="entry__sub">
              <span>{a.itemCosts.map((c) => {
                const short = !isWorking && count(state.inventory, c.item) < c.amount - Math.min(e.costsConsumed, c.amount);
                return <span key={c.item}>{short ? <span className="warn">⚠ </span> : null}{c.item} {Math.min(e.costsConsumed, c.amount)}/{c.amount} </span>;
              })}</span>
              <span className="ink-2">{seconds(remainingSeconds(state, content, e))}</span>
            </div>
            {missing && <div className="entry__sub ink-2">waiting on {missing}</div>}
          </div>
        );
      })}
    </section>
```

```tsx
// src/ui/App.tsx
import { useEffect, useRef } from 'react';
import { scrub, SCRUB_HEAD, SCRUB_ORDER } from '../data/scrub';
import { firstRunnable } from '../engine/queue';
import { ticksPerSecond, ticksToSeconds } from '../engine/time';
import { installDevHandle } from '../state/devHandle';
import { useGame } from '../state/useGame';
import { ChapterPanel } from './ChapterPanel';
import { clock } from './format';
import { HealthBar } from './HealthBar';
import { GearIcon } from './icons';
import { Log } from './Log';
import { Pack } from './Pack';
import { Queue } from './Queue';
import { Rates } from './Rates';
import { SkillsBand } from './SkillsBand';

export function App() {
  const { state, log, dispatch } = useGame(scrub);
  const latest = useRef(state);
  latest.current = state;
  useEffect(() => {
    installDevHandle({
      state: () => latest.current,
      dispatch,
      step: (n) => { for (let i = 0; i < n; i++) dispatch({ type: 'tick' }); },
    });
  }, [dispatch]);

  const working = firstRunnable(state, scrub);
  const runningActionId = working === -1 ? null : state.queue[working]!.actionId;
  const runningSkill = runningActionId ? scrub.actions[runningActionId]!.verb : null;
  const live = state.paused === 'none' && !state.dead;
  const clockNote = !live ? null : state.queue.length === 0 ? 'idle' : working === -1 ? 'waiting' : null;

  return (
    <main className="screen">
      <div className="top">
        <HealthBar health={state.health} max={state.maxHealth} />
        <div className="corner">
          <GearIcon aria-label="settings" className="corner__gear" />
          <span className={`corner__clock${clockNote ? ' corner__clock--dim' : ''}`} aria-label="run clock">
            {clock(ticksToSeconds(state.runTicks))}{clockNote && <small> {clockNote}</small>}
          </span>
          {state.paused === 'none'
            ? <button type="button" className="btn" aria-label="pause" onClick={() => dispatch({ type: 'pause' })}>❚❚</button>
            : <button type="button" className="btn btn--play" aria-label="resume" onClick={() => dispatch({ type: 'resume' })}>▶</button>}
          <span className="visually-hidden" aria-label="tick rate">{ticksPerSecond()} ticks per second</span>
        </div>
      </div>
      <SkillsBand skills={state.skills} runningSkill={runningSkill} />
      <div className="columns">
        <ChapterPanel
          content={scrub} order={SCRUB_ORDER} head={SCRUB_HEAD} state={state} runningActionId={runningActionId}
          onNow={(id) => dispatch({ type: 'queue', actionId: id, front: true })}
          onStop={() => { if (working !== -1) dispatch({ type: 'remove', index: working }); }}
          onQueue={(id) => dispatch({ type: 'queue', actionId: id })}
        />
        <div className="middle">
          <Rates state={state} />
          <Pack state={state} content={scrub} />
          <Log lines={log} content={scrub} />
        </div>
        <Queue state={state} content={scrub} onRemove={(i) => dispatch({ type: 'remove', index: i })} />
      </div>
      {state.dead && <div className="dead">Dead at {clock(ticksToSeconds(state.runTicks))}. Rebirth is a later milestone.</div>}
    </main>
  );
}
```
